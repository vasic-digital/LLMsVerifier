Key Components

Provider Adapter Pattern

Each LLM provider implements a common interface defined in providers/base.go:

type Provider interface {
    // Name returns the provider identifier
    Name() string

    // Discover returns available models
    Discover(ctx context.Context) ([]Model, error)

    // Verify tests a model's capabilities
    Verify(ctx context.Context, model string) (*VerificationResult, error)

    // Complete sends a completion request
    Complete(ctx context.Context, req *CompletionRequest) (*CompletionResponse, error)

    // Stream sends a streaming completion request
    Stream(ctx context.Context, req *CompletionRequest) (<-chan *StreamChunk, error)

    // Health checks provider availability
    Health(ctx context.Context) error
}

Verification Engine

The verification engine in verification/ runs multiple checks against models:

  • AvailabilityCheck - API connectivity
  • AuthenticationCheck - API key validity
  • ResponseCheck - Quality and coherence
  • CodeUnderstandingCheck - "Do you see my code?" test
  • StreamingCheck - Streaming support
  • CapabilityCheck - Feature detection

Scoring System

The scoring engine calculates weighted scores based on:

  • Speed (20%): Latency and throughput metrics
  • Efficiency (20%): Context utilization
  • Cost (15%): Price-to-performance ratio
  • Capability (30%): Feature set and quality
  • Recency (15%): Knowledge freshness

Development Setup

Prerequisites

  • Go 1.24 or later
  • Docker or Podman
  • Make
  • Git

Clone and Build

# Clone the repository
git clone https://github.com/llm-verifier/llm-verifier.git
cd llm-verifier

# Install development tools
make setup

# Build the binary
make build

# Verify installation
./bin/llm-verifier --version

Configuration

# Copy example config
cp llm-verifier/config.yaml.example config.yaml

# Set required environment variables
export OPENAI_API_KEY="sk-..."
export ANTHROPIC_API_KEY="sk-ant-..."
export DB_ENCRYPTION_KEY=$(openssl rand -hex 32)
export JWT_SECRET=$(openssl rand -hex 32)

Run Development Server

# Start with hot reload (requires air)
make dev

# Or run directly
go run ./llm-verifier/cmd server --dev

Adding a Custom Provider

Step 1: Create Provider File

Create a new file at llm-verifier/providers/myprovider.go:

package providers

import (
    "context"
    "fmt"
    "net/http"
)

// MyProvider implements the Provider interface
type MyProvider struct {
    baseURL   string
    apiKey    string
    client    *http.Client
}

// NewMyProvider creates a new MyProvider instance
func NewMyProvider(apiKey string, opts ...Option) *MyProvider {
    p := &MyProvider{
        baseURL: "https://api.myprovider.com/v1",
        apiKey:  apiKey,
        client:  &http.Client{Timeout: 30 * time.Second},
    }
    for _, opt := range opts {
        opt(p)
    }
    return p
}

func (p *MyProvider) Name() string {
    return "myprovider"
}

func (p *MyProvider) Discover(ctx context.Context) ([]Model, error) {
    // Implement model discovery
    return []Model{
        {Name: "my-model-1", Provider: p.Name()},
        {Name: "my-model-2", Provider: p.Name()},
    }, nil
}

func (p *MyProvider) Verify(ctx context.Context, model string) (*VerificationResult, error) {
    // Run verification checks
    result := &VerificationResult{
        Model:    model,
        Provider: p.Name(),
        Checks:   make([]Check, 0),
    }

    // Check availability
    if err := p.Health(ctx); err != nil {
        result.Checks = append(result.Checks, Check{
            Name:   "availability",
            Passed: false,
            Error:  err.Error(),
        })
        return result, nil
    }
    result.Checks = append(result.Checks, Check{Name: "availability", Passed: true})

    // Additional checks...
    return result, nil
}

func (p *MyProvider) Complete(ctx context.Context, req *CompletionRequest) (*CompletionResponse, error) {
    // Implement completion API call
    // ...
}

func (p *MyProvider) Stream(ctx context.Context, req *CompletionRequest) (<-chan *StreamChunk, error) {
    // Implement streaming API call
    // ...
}

func (p *MyProvider) Health(ctx context.Context) error {
    req, err := http.NewRequestWithContext(ctx, "GET", p.baseURL+"/models", nil)
    if err != nil {
        return err
    }
    req.Header.Set("Authorization", "Bearer "+p.apiKey)

    resp, err := p.client.Do(req)
    if err != nil {
        return err
    }
    defer resp.Body.Close()

    if resp.StatusCode != http.StatusOK {
        return fmt.Errorf("health check failed: %d", resp.StatusCode)
    }
    return nil
}

Step 2: Register the Provider

Add registration in llm-verifier/providers/registry.go:

func init() {
    Register("myprovider", func(cfg *Config) (Provider, error) {
        return NewMyProvider(cfg.APIKey), nil
    })
}

Step 3: Add Configuration

Update config.yaml:

llms:
  - name: myprovider
    endpoint: "https://api.myprovider.com/v1"
    api_key: "${MY_PROVIDER_API_KEY}"

Step 4: Write Tests

// llm-verifier/providers/myprovider_test.go
package providers

import (
    "context"
    "testing"
    "github.com/stretchr/testify/assert"
    "github.com/stretchr/testify/require"
)

func TestMyProvider_Name(t *testing.T) {
    p := NewMyProvider("test-key")
    assert.Equal(t, "myprovider", p.Name())
}

func TestMyProvider_Discover(t *testing.T) {
    p := NewMyProvider("test-key")
    models, err := p.Discover(context.Background())
    require.NoError(t, err)
    assert.NotEmpty(t, models)
}

func TestMyProvider_Health(t *testing.T) {
    // Skip in CI without API key
    if os.Getenv("MY_PROVIDER_API_KEY") == "" {
        t.Skip("MY_PROVIDER_API_KEY not set")
    }

    p := NewMyProvider(os.Getenv("MY_PROVIDER_API_KEY"))
    err := p.Health(context.Background())
    assert.NoError(t, err)
}

Verification System

Adding a Custom Check

Create a new check in llm-verifier/verification/checks/:

// llm-verifier/verification/checks/custom_check.go
package checks

import (
    "context"
)

type CustomCheck struct {
    name string
}

func NewCustomCheck() *CustomCheck {
    return &CustomCheck{name: "custom-check"}
}

func (c *CustomCheck) Name() string {
    return c.name
}

func (c *CustomCheck) Run(ctx context.Context, provider Provider, model string) (*CheckResult, error) {
    result := &CheckResult{
        Name:   c.name,
        Passed: true,
    }

    // Implement your check logic
    resp, err := provider.Complete(ctx, &CompletionRequest{
        Model: model,
        Messages: []Message{
            {Role: "user", Content: "Test prompt for custom check"},
        },
    })
    if err != nil {
        result.Passed = false
        result.Error = err.Error()
        return result, nil
    }

    // Validate response
    if len(resp.Content) < 10 {
        result.Passed = false
        result.Error = "Response too short"
    }

    return result, nil
}

Scoring Engine

Modifying Score Weights

Adjust weights in configuration:

scoring:
  weights:
    speed: 0.15       # Reduce speed weight
    efficiency: 0.20
    cost: 0.20        # Increase cost weight
    capability: 0.30
    recency: 0.15

Adding a Score Component

// llm-verifier/verification/scoring/reliability.go
package scoring

type ReliabilityScorer struct{}

func NewReliabilityScorer() *ReliabilityScorer {
    return &ReliabilityScorer{}
}

func (s *ReliabilityScorer) Name() string {
    return "reliability"
}

func (s *ReliabilityScorer) Calculate(metrics *Metrics) float64 {
    // Calculate reliability score based on uptime, error rates, etc.
    score := 10.0

    // Penalize for errors
    if metrics.ErrorRate > 0 {
        score -= metrics.ErrorRate * 20
    }

    // Penalize for timeouts
    if metrics.TimeoutRate > 0 {
        score -= metrics.TimeoutRate * 30
    }

    if score < 0 {
        score = 0
    }
    return score
}

Contribution Guidelines

Before Contributing

  1. Check existing issues and PRs for similar work
  2. Open an issue to discuss major changes
  3. Fork the repository
  4. Create a feature branch from main

Requirements

  • All tests must pass: make test
  • Code must be formatted: make fmt
  • Linting must pass: make lint
  • Security scan must pass: make security
  • New features require tests
  • Documentation for public APIs

Pull Request Process

  1. Create a branch: git checkout -b feature/my-feature
  2. Make changes and commit with descriptive messages
  3. Push to your fork: git push origin feature/my-feature
  4. Open a PR against main
  5. Address review feedback
  6. Squash commits if requested

Code Style

Go Style Guide

  • Follow Effective Go
  • Use gofmt for formatting
  • Export only necessary identifiers
  • Document all exported functions
  • Handle all errors explicitly
  • Use context for cancellation

Naming Conventions

// Good
type ProviderConfig struct {}
func NewProvider(cfg *ProviderConfig) *Provider
func (p *Provider) Verify(ctx context.Context) error

// Avoid
type providerconfig struct {}  // Not exported
func createProvider(cfg *Config) *Provider  // Use New prefix
func (p *Provider) DoVerify() error  // Avoid Do prefix

Error Handling

// Always wrap errors with context
if err := p.Health(ctx); err != nil {
    return nil, fmt.Errorf("health check failed for %s: %w", p.Name(), err)
}

// Use custom error types for specific cases
type ProviderError struct {
    Provider string
    Code     string
    Message  string
}

func (e *ProviderError) Error() string {
    return fmt.Sprintf("[%s] %s: %s", e.Provider, e.Code, e.Message)
}
